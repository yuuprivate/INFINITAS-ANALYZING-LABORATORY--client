const DWORD processId =
    ProcessFinder::waitForProcess();

Process process;

process.open(processId);

std::uintptr_t baseAddress = 0;

process.getMainModuleBaseAddress(
    baseAddress
);

MemoryReader memoryReader(process);

VersionDetector versionDetector(
    memoryReader
);

std::string detectedVersion;

versionDetector.detect(
    baseAddress,
    detectedVersion
);

OffsetManager offsetManager;

offsetManager.load(
    "offsets.txt",
    baseAddress
);

if (detectedVersion != offsetManager.version())
{
    return 1;
}
